\begin{question}
	Show that:
\end{question}

\begin{subquestion}
	$(\alpha + \beta) + \lambda = \alpha + (\beta + \lambda)$ for all
	$\alpha, \beta, \lambda \in \mathbb{C}$.
\end{subquestion}

\begin{solution}
	Let $\alpha = \alpha_1 + \alpha_2{i}, \beta = \beta_1 + \beta_2{i}, \lambda = \lambda_1 + \lambda_2{i}$, where $\alpha_1, \alpha_2, \beta_1, \beta_2, \lambda_1, \lambda_2 \in \mathbb{R}$. We therefore have that:

	\begin{alignat*}{2}
		(\alpha + \beta) + \lambda &= (\alpha_1 + \alpha_2{i} + \beta_1 + \beta_2{i}) + \lambda_1 + \lambda_2{i} \qquad && \comment{(By definition of $\alpha, \beta, \lambda$)} \\\\
		&= (\alpha_1 + \beta_1 + \alpha_2{i} + \beta_2{i}) + \lambda_1 + \lambda_2{i} \qquad && \comment{(By commutativity of addition)} \\\\
		&= (\alpha_1 + \beta_1 + (\alpha_2 + \beta_2)i) + \lambda_1 + \lambda_2{i} \qquad && \comment{(By distributivity of multiplication over addition)} \\\\
		&= (\alpha_1 + \beta_1 + (\alpha_2 + \beta_2)i) \\\\
		&= \alpha_1 + \beta_1 + (\alpha_2 + \beta_2)i + \lambda_1 + \lambda_2{i} \qquad && \comment{(By definition of addition over $\mathbb{C}$)} \\\\
		&= ((\alpha_1 + \beta_1) + \lambda_1) + ((\alpha_2 + \beta_2) + \lambda_2){i} \qquad && \comment{(By definition of addition over $\mathbb{C}$)} \\\\
		&= (\alpha_1 + (\beta_1 + \lambda_1)) + (\alpha_2 + (\beta_2 + \lambda_2)){i} \qquad && \comment{(By associativity of addition)} \\\\
		&= \alpha_1 + (\beta_1 + \lambda_1) + \alpha_2{i} + (\beta_2 + \lambda_2){i} \qquad && \comment{(By distributivity)} \\\\
		&= \alpha_1 + \alpha_2{i} + (\beta_1 + \lambda_1) + (\beta_2 + \lambda_2){i} \qquad && \comment{(By commutativity of addition)} \\\\
		&= \alpha + (\beta + \lambda) \qquad && \comment{By definition of $\alpha, \beta, \lambda$} \\\\
	\end{alignat*}
\end{solution}

\begin{subquestion}
	$(\alpha + \beta) + \lambda = \alpha + (\beta + \lambda)$ for all
	$\alpha, \beta, \lambda \in \mathbb{Z} / 2\mathbb{Z}$.
\end{subquestion}

\begin{solution}
	In the case that $\alpha, \beta, \lambda \in \mathbb{Z} / 2\mathbb{Z}$, we have that:

	\begin{gather*}
		(\alpha + \beta) + \lambda = \alpha + (\beta + \lambda)
	\end{gather*}

	Since $\mathbb{Z} / 2\mathbb{Z}$ uses regular integer addition with the exception that $1 + 1 \coloneq 0$. Since this is still associative, $(\alpha + \beta) + \lambda = \alpha + (\beta + \lambda)$ for all $\alpha, \beta, \lambda \in \mathbb{Z} / 2\mathbb{Z}$.
\end{solution}

\begin{question}
	Show that:
\end{question}

\begin{subquestion}
	$\lambda(\alpha + \beta) = \lambda\alpha + \lambda\beta$ for all
	$\lambda, \alpha, \beta \in \mathbb{C}$.
\end{subquestion}

\begin{solution}
	As earlier, letting $\alpha = \alpha_1 + \alpha_2{i}, \beta = \beta_1 + \beta_2{i}$, and $\lambda = \lambda_1 + \lambda_2{i}$ respectively, we have that:

	\begin{align*}
		\lambda(\alpha + \beta) &= (\lambda_1 + \lambda_2{i})((\alpha_1 + \alpha_2{i}) + (\beta_1 + \beta_2{i})) \\\\
		\qquad & \text{(By definition of $\alpha, \beta, \lambda$)} \\\\
		&= \lambda_1((\alpha_1 + \alpha_2{i}) + (\beta_1 + \beta_2{i})) + \lambda_2{i}((\alpha_1 + \alpha_2{i}) + (\beta_1 + \beta_2{i})) \\\\
		\qquad & \text{(By distributivity of multiplication over addition)} \\\\
		&= \lambda_1(\alpha_1 + \alpha_2{i}) + \lambda_1(\beta_1 + \beta_2{i}) + \lambda_2{i}(\alpha_1 + \alpha_2{i}) + \lambda_2{i}(\beta_1 + \beta_2{i}) \\\\
		\qquad & \text{(By distributivity of multiplication over addition)} \\\\
		&= \lambda_1(\alpha_1 + \alpha_2{i}) + \lambda_2{i}(\alpha_1 + \alpha_2{i}) + \lambda_1(\beta_1 + \beta_2{i}) + \lambda_2{i}(\beta_1 + \beta_2{i}) \\\\
		\qquad & \text{(By commutativity of addition)} \\\\
		&= (\lambda_1 + \lambda_2{i})(\alpha_1 + \alpha_2{i}) + (\lambda_1 + \lambda_2{i})(\beta_1 + \beta_2{i}) \\\\
		\qquad & \text{(By distributivity of multiplication over addition)} \\\\
		&= \lambda\alpha + \lambda\beta \\\\
		\qquad & \text{(By definition of $\alpha, \beta, \lambda$)} \\\\
	\end{align*}
\end{solution}

\begin{subquestion}
	$\lambda(\alpha + \beta) = \lambda\alpha + \lambda\beta$ for all
	$\lambda, \alpha, \beta \in \mathbb{Z} / 2\mathbb{Z}$.
\end{subquestion}

\begin{solution}
	Since multiplication is defined as usual, on $\mathbb{Z} / 2\mathbb{Z}$, and integer multiplication is distributive, we have that

	\begin{gather*}
		\lambda(\alpha + \beta) = \lambda\alpha + \lambda\beta
	\end{gather*}

	for all $\lambda, \alpha, \beta \in \mathbb{Z} / 2\mathbb{Z}$.
\end{solution}

\begin{question}
	Show that:
\end{question}

\begin{subquestion}
	For every $\alpha \in \mathbb{C}$ with $\alpha \neq 0$, there exists a
	unique $\beta \in \mathbb{C}$ such that $\alpha\beta = 1$.
\end{subquestion}

\begin{solution}
	Let $\alpha = a + bi$, where $\alpha \neq 0$. I assert that $\beta = \frac{1}{a + bi}$ would satisfy $\alpha\beta = 1$, to prove $\beta \in \mathbb{C}$, consider that, since $\alpha \neq 0$, $a \neq 0 \vee b \neq 0$ and thus $\frac{1}{a + bi}$ is defined. Therefore, we can see that:

	\begin{align*}
		\frac{1}{a + bi} &= \frac{1}{a + bi} \cdot 1 \qquad && \comment{(By multiplicative identity)} \\\\
		&= \frac{1}{a + bi} \cdot {a - bi}{a - bi} \qquad && \comment{(By multiplicative inverse)} \\\\
		&= \frac{a - bi}{a^2 - (bi)^2} \qquad && \comment{(By usual rules of multiplication)} \\\\
		&= \frac{a - bi}{a^2 + b^2} \qquad && \comment{(By definition of $i$. Note that $\alpha \neq 0 \implies a \neq 0 \vee b \neq 0 \implies a^2 + b^2 > 0$)} \\\\
		&- \frac{a}{a^2 + b^2} + \frac{-b}{a^2 + b^2}i \qquad && \comment{(By usual rules of addition)} \\\\
	\end{align*}

	Thus, since $\frac{a}{a^2 + b^2}, \frac{-b}{a^2 + b^2} \in \mathbb{R}$, $\frac{1}{a + bi} \in \mathbb{C}$ and therefore $\beta = \frac{1}{a + bi}$ satisfies $b \in \mathbb{C}$ such that $\alpha\beta = 1$ when $\alpha \neq 0$. \\

	To prove that the multiplicative inverse of $\alpha$ is unique, simply consider the case that $\alpha\beta_1 = 1$ and $\alpha\beta_2 = 1$ where $\beta_1, \beta_2 \in \mathbb{C}$. Then, we have that:

	\begin{align*}
		\beta_1 &= 1 \cdot \beta_1 \qquad && \comment{(By multiplicative identity)} \\\\
		&= (\alpha\beta_2)\beta_1 \qquad && \comment{(By definition of $\beta_2$)} \\\\
		&= (\beta_2\alpha)\beta_1 \qquad && \comment{(By commutativity of multiplication)} \\\\
		&= \beta_2(\alpha\beta_1) \qquad && \comment{(By associativity of multiplication)} \\\\
		&= \beta_2 \cdot 1 \qquad && \comment{(By definition of $\beta_1$)} \\\\
		&= \beta_2 \qquad && \comment{(By multiplicative identity)}
	\end{align*}

	Thus, $\beta_1 = \beta_2$ and therefore the multiplicative inverse of $\alpha$ must be unique.
\end{solution}

\begin{subquestion}
	For every $\alpha \in \mathbb{Z} / 2\mathbb{Z}$ with
	$\alpha \neq 0$, there exists a unique $\beta \in \mathbb{Z} / 2\mathbb{Z}$ such that
	$\alpha\beta = 1$.
\end{subquestion}

\begin{solution}
	If $\alpha \in \mathbb{Z} / 2\mathbb{Z}$, then $\alpha = 0 \vee \alpha = 1$. Thus, if $\alpha \neq 0$, then $\alpha = 1$. Thus, selecting $\beta = 1$, we have that $\beta \in \mathbb{Z} / 2\mathbb{Z}$ and:

	\begin{align*}
		\alpha\beta &= 1 \cdot 1 \\
		&= 1 \\
	\end{align*}

	Note that only $\beta = 1$ can possibly be the multiplicative inverse of such an $\alpha$ since, if $\beta = 0$:

	\begin{align*}
		\alpha\beta &= 1 \cdot 0 \\
		&= 0 \\
	\end{align*}

	Thus, the multiplicative inverse in of any non-zero $\alpha$ in $\mathbb{Z} / 2\mathbb{Z}$ must be unique.
\end{solution}

\begin{question}
	Show that $(ab)x = a(bx)$ for all $x \in \mathbb{F}^n$, where
	$\mathbb{F}$ is any field.
\end{question}

\begin{solution}
	Let $x = (x_1, x_2, \ldots, x_n)$. We therefore have that:

	\begin{alignat*}{2}
		(ab)x &= (ab)(x_1, x_2, \ldots, x_n) &&\qquad \comment{(By definition of $x$)} \\\\
		&= ((ab)x_1, (ab)x_2, \ldots, (ab)x_n) &&\qquad \comment{(By definition of scalar multiplication over $n$-tuples)} \\\\
		&= (a(bx_1), a(bx_2), \ldots a(bx_n)) &&\qquad \comment{(By associativity of multiplication)} \\\\
		&= a(bx_1, bx_2, \ldots bx_n) &&\qquad \comment{(By definition of scalar multiplication over $n$-tuples)} \\\\
		&= a(bx) &&\qquad \comment{(By definition of $x$ and scalar multiplication over $n$-tuples)} \\\\
	\end{alignat*}
\end{solution}

\begin{question}
	Show that $(\alpha + \beta)x = \alpha{x} + \beta{x}$ for all $\alpha, \beta \in
	\mathbb{F}$ and all $x \in \mathbb{F}^n$, where $\mathbb{F}$ is any
	field.
\end{question}

\begin{solution}
	Let $x = (x_1, x_2, \ldots, x_n)$. We therefore have that:

	\begin{alignat*}{2}
		(\alpha + \beta)x &= (\alpha + \beta)(x_1, x_2, \ldots, x_n) \qquad && \comment{(By definition of $x$)} \\\\
		&= ((\alpha + \beta)x_1, (\alpha + \beta)x_2, \ldots, (\alpha + \beta)x_n) \qquad && \comment{(By definition of scalar multiplication over $n$-tuples)} \\\\
		&= (\alpha{x_1} + \beta{x_1}, \alpha{x_2} + \beta{x_2}, \ldots, \alpha{x_n} + \beta{x_n}) \qquad && \comment{(By distributive property)} \\\\
		&= (\alpha{x_1}, \alpha{x_2}, \ldots, \alpha{x_n}) + (\beta{x_1}, \beta{x_2}, \ldots, \beta{x_n}) \qquad && \comment{(By definition of addition over $n$-tuples)} \\\\
		&= \alpha(x_1, x_2, \ldots, x_n) + \beta(x_1, x_2, \ldots, x_n) \qquad && \comment{(By definition of scalar multiplication over $n$-tuples)} \\\\
		&= \alpha{x} + \beta{x} \qquad && \comment{(By definition of $x$)} \\\\
	\end{alignat*}
\end{solution}
